def map_input(field_name, value):
    """Map input value to encoded number using predefined mappings"""
    return mappings[field_name][value]

def user_input_form():
    st.sidebar.header("Enter Patient Data")
    inputs = {}
    
    inputs['age'] = st.sidebar.number_input("Age")
    inputs['sex'] = map_input('sex', st.sidebar.selectbox("Sex", list(mappings['sex'].keys())))
    inputs['bmi_curc'] = map_input('bmi_curc', st.sidebar.selectbox("BMI", list(mappings['bmi_curc'].keys())))
    inputs['cig_stat'] = map_input('cig_stat', st.sidebar.selectbox("Smoking Status", list(mappings['cig_stat'].keys())))
    inputs['pack_years'] = st.sidebar.number_input("Pack Years", min_value=0.0, value=0.0)
    inputs['ph_any_trial'] = map_input('ph_any_trial', st.sidebar.selectbox("Clinical Trial", ['No', 'Yes']))
    inputs['diabetes_f'] = map_input('diabetes_f', st.sidebar.selectbox("Diabetes", ['No', 'Yes']))
    inputs['hyperten_f'] = map_input('hyperten_f', st.sidebar.selectbox("Hypertension", ['No', 'Yes']))
    inputs['emphys_f'] = map_input('emphys_f', st.sidebar.selectbox("Emphysema", ['No', 'Yes']))
    inputs['bronchit_f'] = map_input('bronchit_f', st.sidebar.selectbox("Bronchitis", ['No', 'Yes']))
    inputs['hearta_f'] = map_input('hearta_f', st.sidebar.selectbox("Heart Disease", ['No', 'Yes']))
    
    inputs['proc_numl'] = map_input('proc_numl', st.sidebar.selectbox("Procedure", list(mappings['proc_numl'].keys())))
    inputs['del_invas_cat'] = map_input('del_invas_cat', st.sidebar.selectbox("Diagnostic Method", list(mappings['del_invas_cat'].keys())))

    inputs['biopllink0'] = map_input('biopllink0', st.sidebar.selectbox("Biopsy Linked", ['No', 'Yes']))
    inputs['reasfolll'] = map_input('reasfolll', st.sidebar.selectbox("Follow-up Reason", ['No', 'Yes']))
    
    inputs['lung_stage'] = map_input('lung_stage', st.sidebar.selectbox("Cancer Stage", list(mappings['lung_stage'].keys())))
    inputs['lung_clinstage'] = map_input('lung_clinstage', st.sidebar.selectbox("Clinical Stage", list(mappings['lung_clinstage'].keys())))
    inputs['lung_stage_t'] = map_input('lung_stage_t', st.sidebar.selectbox("T Stage", list(mappings['lung_stage_t'].keys())))

# Main app
st.title("Lung Cancer Complication Predictor")
st.write("Predicts complication severity, type, and treatment timing")

# Get inputs
input_df = user_input_form()

# Prediction logic
if st.button("Predict"):
    try:
        # Convert to float for model compatibility
        input_processed = input_df.astype(float)
        
        # Make predictions
        pred_catl = model_ctype_catl.predict(input_processed)
        pred_ctypel = model_ctypel.predict(input_processed)
        pred_gap = model_comp_gap_category.predict(input_processed)
        
        # Get probabilities for ctypel
        pred_ctypel_proba = model_ctypel.predict_proba(input_processed)[0]
        top_5_indices = np.argsort(pred_ctypel_proba)[-5:][::-1]
        top_5 = [(list(ctypel_mapping.keys())[idx], pred_ctypel_proba[idx]) for idx in top_5_indices]

        # Display results
        st.subheader("Predictions:")
        st.write(f"**Severity:** {ctype_catl_mapping[pred_catl[0]]}")
        st.write(f"**Complication Type:** {list(ctypel_mapping.keys())[pred_ctypel[0]]}")
        st.write(f"**Treatment Timing:** {comp_gap_category_mapping[pred_gap[0]]}")
        
        st.subheader("Top 5 Complication Probabilities:")
        for name, prob in top_5:
            st.write(f"{name}: {prob*100:.2f}%")
            
    except Exception as e:
        st.error(f"Prediction error: {str(e)}")
